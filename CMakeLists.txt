cmake_minimum_required(VERSION 3.20)
project(AL_STUDIO VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "AL STUDIO MVP currently targets Windows.")
endif()
add_executable(AL-STUDIO WIN32 src/main.cpp)
target_compile_features(AL-STUDIO PRIVATE cxx_std_17)
target_compile_definitions(AL-STUDIO PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(AL-STUDIO PRIVATE winmm comdlg32)
if(MSVC)
  target_compile_options(AL-STUDIO PRIVATE /W4 /permissive-)
endif()
